
#include <stdio.h>
#include <string.h>

int main() {
    char data[50], div[20];
    int i, j, n, m;

    printf("Enter data bits: ");
    scanf("%s", data);
    printf("Enter divisor bits: ");
    scanf("%s", div);

    n = strlen(data);
    m = strlen(div);

    for (i = 0; i < m - 1; i++)
        data[n + i] = '0';
    data[n + m - 1] = '\0';

    for (i = 0; i < n; i++) {
        if (data[i] == '1') {
            for (j = 0; j < m; j++)
                data[i + j] =
                (data[i + j] == div[j]) ? '0' : '1';
        }
    }

    printf("CRC: ");
    for (i = n; i < n + m - 1; i++)
        printf("%c", data[i]);

    printf("\n");
    return 0;
}

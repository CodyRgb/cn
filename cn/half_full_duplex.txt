1. Server (server.c)
#include <stdio.h>
#include <unistd.h>
#include <string.h>
#include <arpa/inet.h>

int main() {
    int s, c;
    char b[100] = {0};
    struct sockaddr_in a = {0};

    s = socket(AF_INET, SOCK_STREAM, 0);
    a.sin_family = AF_INET;
    a.sin_port = htons(8082);
    a.sin_addr.s_addr = INADDR_ANY;

    bind(s, (struct sockaddr*)&a, sizeof(a));
    listen(s, 1);
    printf("Waiting for client...\n");

    c = accept(s, NULL, NULL);
    read(c, b, 99);
    printf("Client: %s\n", b);

    write(c, "Hello Client", 12);

    close(c);
    close(s);
    return 0;
}


2. Client (client.c)
#include <stdio.h>
#include <unistd.h>
#include <string.h>
#include <arpa/inet.h>

int main() {
    int s;
    char b[100] = {0};
    struct sockaddr_in a = {0};

    s = socket(AF_INET, SOCK_STREAM, 0);
    a.sin_family = AF_INET;
    a.sin_port = htons(8082);
    inet_pton(AF_INET, "127.0.0.1", &a.sin_addr);

    connect(s, (struct sockaddr*)&a, sizeof(a));

    write(s, "Hello Server", 12);
    read(s, b, 99);
    printf("Server: %s\n", b);

    close(s);
    return 0;
}

How to run these programs in Ubuntu
For each experiment:
1. Save the server and client code in separate files.
2. Compile both:
   gcc server.c -o server
   gcc client.c -o client
3. Open Terminal 1 and run:
   ./server
4. Open Terminal 2 and run:
   ./client

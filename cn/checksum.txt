#include <stdio.h>

int main() {
    int a, b, sum, checksum;

    printf("Enter two 4-bit decimal values: ");
    scanf("%d%d", &a, &b);

    sum = a + b;
    sum = (sum & 15) + (sum >> 4);
    checksum = (~sum) & 15;

    printf("Checksum in binary: ");
    for (int i = 3; i >= 0; i--)
        printf("%d", (checksum >> i) & 1);

    printf("\n");
    return 0;
}

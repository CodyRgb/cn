#include <stdio.h>
#define INF 999

int main() {
    int a[10][10], d[10], vis[10] = {0};
    int n, src, i, j, u, min;

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    printf("Enter cost matrix (999 for no link):\n");
    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++)
            scanf("%d", &a[i][j]);

    printf("Enter source node (0 to %d): ", n - 1);
    scanf("%d", &src);

    for (i = 0; i < n; i++)
        d[i] = a[src][i];
    d[src] = 0;

    for (i = 0; i < n; i++) {
        min = INF;
        u = -1;

        for (j = 0; j < n; j++)
            if (!vis[j] && d[j] < min)
                min = d[j], u = j;

        if (u == -1) break;
        vis[u] = 1;

        for (j = 0; j < n; j++)
            if (!vis[j] && a[u][j] < INF &&
                d[u] + a[u][j] < d[j])
                d[j] = d[u] + a[u][j];
    }

    for (i = 0; i < n; i++)
        printf("Distance to %d = %d\n", i, d[i]);

    return 0;
}

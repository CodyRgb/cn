#include <stdio.h>

int main() {
    int d[8], i, p1, p2, p4;

    printf("Enter 4 data bits: ");
    scanf("%1d%1d%1d%1d",
          &d[3], &d[5], &d[6], &d[7]);

    d[1] = d[3] ^ d[5] ^ d[7];
    d[2] = d[3] ^ d[6] ^ d[7];
    d[4] = d[5] ^ d[6] ^ d[7];

    printf("Hamming code: ");
    for (i = 1; i <= 7; i++)
        printf("%d", d[i]);

    printf("\n");
    return 0;
}
